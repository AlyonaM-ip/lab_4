**Часть 1. Разработка модуля ядра для инспекции таблиц страниц**  
**1. Исходный код модуля ядра (**`pgtable_walk.c`**)**  
Ниже представлен полный исходный код разработанного загружаемого модуля ядра (LKM), который принимает на вход идентификатор процесса (`PID`) и целевой виртуальный адрес (`vaddr`), после чего выполняет итеративный аппаратный обход (Page-Table Walk) многоуровневой структуры таблиц страниц x86-64.  
```  
#include <linux/module.h>  
 #include <linux/kernel.h>  
 #include <linux/init.h>  
 #include <linux/pid.h>  
 #include <linux/sched.h>  
 #include <linux/mm.h>  
 #include <linux/highmem.h>  
 #include <asm/pgtable.h>  
   
 MODULE_LICENSE("GPL");  
 MODULE_AUTHOR("AI Collaborator & Student");  
 MODULE_DESCRIPTION("Kernel module for explicit x86-64 page table walk");  
 MODULE_VERSION("1.0");  
   
 // Параметры модуля, передаваемые при загрузке (insmod)  
 static int target_pid = 0;  
 static unsigned long vaddr = 0;  
   
 module_param(target_pid, int, 0644);  
 MODULE_PARM_DESC(target_pid, "Target process PID");  
   
 module_param(vaddr, ulong, 0644);  
 MODULE_PARM_DESC(vaddr, "Virtual address to translate");  
   
 static int __init pgtable_walk_init(void)  
 {  
     struct pid *pid_struct;  
     struct task_struct *task;  
     struct mm_struct *mm;  
       
     pgd_t *pgd;  
     p4d_t *p4d;  
     pud_t *pud;  
     pmd_t *pmd;  
     pte_t *pte;  
     spinlock_t *ptl;  
     unsigned long pfn;  
     unsigned long offset = vaddr & (PAGE_SIZE - 1); // Младшие 12 бит (смещение)  
   
     printk(struct_name, KERN_INFO "=== pgtable_walk: Start translation for PID %d, VA 0x%lx ===\n", target_pid, vaddr);  
   
     // 1. Поиск структуры pid по номеру процесса  
     pid_struct = find_vpid(target_pid);  
     if (!pid_struct) {  
         printk(KERN_ERR "pgtable_walk: Invalid PID %d\n", target_pid);  
         return -EINVAL;  
     }  
   
     // 2. Получение task_struct процесса из структуры pid  
     task = pid_task(pid_struct, PIDTYPE_PID);  
     if (!task) {  
         printk(KERN_ERR "pgtable_walk: Failed to get task_struct\n");  
         return -EINVAL;  
     }  
   
     // 3. Получение контекста памяти процесса (mm_struct)  
     mm = task->mm;  
     if (!mm) {  
         printk(KERN_ERR "pgtable_walk: Process has no mm_struct (kernel thread?)\n");  
         return -EINVAL;  
     }  
   
     // 4. ЗАХВАТ БЛОКИРОВКИ: Замораживаем карту виртуальной памяти процесса.  
     // Это исключает гонку данных, если процесс параллельно вызовет munmap() или free().  
     mmap_read_lock(mm);  
   
     // 5. Шаг 1: Уровень PGD (Page Global Directory) — Корень иерархии  
     pgd = pgd_offset(mm, vaddr);  
     if (pgd_none(*pgd) || !pgd_present(*pgd)) {  
         printk(KERN_WARNING "pgtable_walk: PGD entry not present for VA 0x%lx\n", vaddr);  
         goto out_unlock;  
     }  
     printk(KERN_INFO "pgtable_walk: PGD index = %lu, PGD val = 0x%lx\n", pgd_index(vaddr), pgd_val(*pgd));  
   
     // 6. Шаг 2: Уровень P4D (В x86-64 схлопнут (folded), но обязателен для generic API Linux)  
     p4d = p4d_offset(pgd, vaddr);  
     if (p4d_none(*p4d) || !p4d_present(*p4d)) {  
         printk(KERN_WARNING "pgtable_walk: P4D entry not present\n");  
         goto out_unlock;  
     }  
   
     // 7. Шаг 3: Уровень PUD (Page Upper Directory)  
     pud = pud_offset(p4d, vaddr);  
     if (pud_none(*pud) || !pud_present(*pud)) {  
         printk(KERN_WARNING "pgtable_walk: PUD entry not present\n");  
         goto out_unlock;  
     }  
     printk(KERN_INFO "pgtable_walk: PUD index = %lu, PUD val = 0x%lx\n", pud_index(vaddr), pud_val(*pud));  
   
     // 8. Шаг 4: Уровень PMD (Page Middle Directory)  
     pmd = pmd_offset(pud, vaddr);  
     if (pmd_none(*pmd) || !pmd_present(*pmd)) {  
         printk(KERN_WARNING "pgtable_walk: PMD entry not present\n");  
         goto out_unlock;  
     }  
       
     // Проверка флага больших страниц (_PAGE_PSE). Если он взведен, обход до PTE не идет.  
     if (pmd_large(*pmd)) {  
         printk(KERN_INFO "pgtable_walk: Detected Huge Page (2 MiB) at PMD level\n");  
         pfn = pmd_pfn(*pmd);  
         printk(KERN_INFO "pgtable_walk: Final Physical Address = 0x%lx\n", (pfn << PAGE_SHIFT) | (vaddr & ((1UL << PMD_SHIFT) - 1)));  
         goto out_unlock;  
     }  
     printk(KERN_INFO "pgtable_walk: PMD index = %lu, PMD val = 0x%lx\n", pmd_index(vaddr), pmd_val(*pmd));  
   
     // 9. Шаг 5: Уровень PTE (Page Table Entry) — Финальная таблица страниц  
     // Используем pte_offset_map_lock для безопасного маппинга страницы таблицы PTE и ее блокировки  
     pte = pte_offset_map_lock(mm, pmd, vaddr, &ptl);  
     if (!pte) {  
         printk(KERN_WARNING "pgtable_walk: Failed to map PTE\n");  
         goto out_unlock;  
     }  
   
     if (pte_none(*pte) || !pte_present(*pte)) {  
         printk(KERN_WARNING "pgtable_walk: PTE entry not present\n");  
         pte_unmap_unlock(pte, ptl);  
         goto out_unlock;  
     }  
     printk(KERN_INFO "pgtable_walk: PTE index = %lu, PTE val = 0x%lx\n", pte_index(vaddr), pte_val(*pte));  
   
     // 10. Извлечение номера физического кадра (PFN) и расчет физического адреса (PA)  
     pfn = pte_pfn(*pte);  
     printk(KERN_INFO "pgtable_walk: Found PFN = 0x%lx\n", pfn);  
     printk(KERN_INFO "pgtable_walk: Final Physical Address = 0x%lx\n", (pfn << PAGE_SHIFT) | offset);  
   
     // Освобождаем низкоуровневую блокировку PTE  
     pte_unmap_unlock(pte, ptl);  
   
 out_unlock:  
     // 11. ОСВОБОЖДЕНИЕ БЛОКИРОВКИ: Возвращаем процессу возможность менять память  
     mmap_read_unlock(mm);  
     return 0;  
 }  
   
 static void __exit pgtable_walk_exit(void)  
 {  
     printk(KERN_INFO "=== pgtable_walk: Unloaded ===\n");  
 }  
   
 module_init(pgtable_walk_init);  
 module_exit(pgtable_exit);  
   
```

**Часть 2. Независимая проверка через /proc/\[pid\]/pagemap и визуализация**  
**1. Исходный код скрипта** `pagemap_reader.py`  
Userspace-скрипт предназначен для независимой валидации трансляции адресов без использования модулей ядра. Он парсит виртуальные регионы из `/proc/[pid]/maps`, затем рассчитывает смещение в байтах для каждой виртуальной страницы и вычитывает 64-битное состояние (запись PTE) напрямую из файла `/proc/[pid]/pagemap`.  
```  
#!/usr/bin/env python3  
 import sys  
 import os  
 import struct  
   
 PAGE_SIZE = 4096  
   
 def read_maps(pid):  
     """Парсит файл /proc/[pid]/maps и возвращает список виртуальных регионов"""  
     regions = []  
     maps_path = f"/proc/{pid}/maps"  
     if not os.path.exists(maps_path):  
         print(f"Ошибка: Процесс {pid} не найден.")  
         sys.exit(1)  
           
     with open(maps_path, 'r') as f:  
         for line in f:  
             parts = line.split()  
             if len(parts) >= 2:  
                 addr_range = parts[0]  
                 start_str, end_str = addr_range.split('-')  
                 start = int(start_str, 16)  
                 end = int(end_str, 16)  
                 name = parts[-1] if parts[-1].startswith('[') or parts[-1].startswith('/') else 'anonymous'  
                 regions.append((start, end, name))  
     return regions  
   
 def get_pagemap_entry(pid, vaddr):  
     """Считывает 64-битную запись из /proc/[pid]/pagemap для заданного vaddr"""  
     pagemap_path = f"/proc/{pid}/pagemap"  
     page_index = vaddr // PAGE_SIZE  
     offset = page_index * 8  # Каждая запись занимает 8 байт (64 бита)  
       
     try:  
         with open(pagemap_path, 'rb') as f:  
             f.seek(offset)  
             entry_bytes = f.read(8)  
             if len(entry_bytes) < 8:  
                 return None  
             return struct.unpack('Q', entry_bytes)[0]  
     except PermissionError:  
         print("Ошибка: Требуются права суперпользователя (sudo) для чтения pagemap.")  
         sys.exit(1)  
   
 def main():  
     if len(sys.argv) < 2:  
         print("Использование: sudo ./pagemap_reader.py <PID>")  
         sys.exit(1)  
           
     pid = int(sys.argv[1])  
     regions = read_maps(pid)  
       
     print(f"=== Результаты независимой проверки для PID {pid} ===")  
     print(f"{'Сегмент':<15} | {'Вирт. Адрес (VA)':<18} | {'Present':<7} | {'Физический Кадр (PFN)':<20}")  
     print("-" * 70)  
       
     # Проверим три ключевых адреса из предыдущего теста  
     test_addresses = {  
         "Код (.text)": 0x62ef7190b000,  
         "Куча (heap)": 0x62efa9153000,  
         "Стек (stack)": 0x7ffdb9b85000  
     }  
       
     for name, vaddr in test_addresses.items():  
         data = get_pagemap_entry(pid, vaddr)  
         if data is None:  
             print(f"{name:<15} | 0x{vaddr:012x} | Error   | Не удалось прочесть")  
             continue  
               
         # Распаковка бит по спецификации ядра Linux  
         present = (data >> 63) & 1  
         swapped = (data >> 62) & 1  
         pfn = data & ((1 << 55) - 1) # Биты 0-54 хранят PFN  
           
         if present:  
             pfn_str = f"0x{pfn:x}"  
         elif swapped:  
             pfn_str = "Swapped out"  
         else:  
             pfn_str = "None (Demand-zero)"  
               
         print(f"{name:<15} | 0x{vaddr:012x} | {present:<7} | {pfn_str:<20}")  
   
 if __name__ == "__main__":  
     main()  
   
```

**2. Объяснение результатов**  
***А. Как работают потоки (****`pthread_create`****)***  
Когда программа создает поток, Linux использует системный вызов `clone()` с флагом **`CLONE_VM`**.  
- **Что это значит:** Этот флаг приказывает новому потоку пользоваться **той же самой таблицей страниц**, что и у главного потока.  
- **Результат:** У них абсолютно общая память. Если один поток читает или меняет переменную, это происходит в одной и той же физической ячейке RAM. Поэтому номера физических кадров (PFN) у них **всегда одинаковые**.  
***Б. Как работают процессы (****`fork()`****)***  
Когда программа создает новый процесс через `fork()`, он должен быть полностью изолирован от родителя. Но копировать гигабайты памяти сразу — это долго. Поэтому Linux хитрит с помощью механизма **`Copy-on-Write` (Копирование при записи)**:  
1. **До записи (Чтение):** Linux копирует только «путеводитель» (таблицу страниц) для нового процесса, а сами физические кадры оставляет общими. Чтобы процессы не испортили данные друг друга, Linux временно запрещает в них писать (делает их Read-Only). Пока процессы просто читают переменную — их виртуальные адреса ведут в один и тот же кадр. **PFN совпадает**.  
2. **После записи:** Как только потомок пытается изменить переменную, процессор видит запрет на запись и бьет тревогу (происходит `Page Fault`). Ядро Linux перехватывает управление, быстро берет **новый пустой кадр памяти**, копирует туда данные старой страницы и разрешает запись. Теперь виртуальный адрес потомка ведет уже в его личную, новую физическую ячейку. **PFN у процессов становятся разными**.  
**Вывод**  
- **Потоки** созданы для быстрой совместной работы, поэтому они живут в одной общей памяти.  
- **Процессы** должны быть изолированы, поэтому Linux дает им общую память только «пока они просто смотрят», но как только начинается запись — мгновенно разделяет их физически.  
**Ответы на контрольные вопросы (Практическое задание №4)**  
**1. Почему для 64-битного адресного пространства линейная таблица страниц физически невозможна на практике, даже если бы производитель процессора теоретически её поддерживал?**  
Требуется очень большой кусок НЕПРЕРЫВНОЙ паямти, а также очень долгий поиск адрессов, даже несмотря на TLB.  
   
 Для стандартной 48-битной адресации x86-64 плоская таблица должна содержать запись для каждой возможной страницы. Математически размер такой таблицы для одного процесса составляет 2^(48 - 12) \* 8  байт = 512 Гигабайт. Она физически не поместится ни в какую оперативную память компьютера, а ведь таких процессов в системе запущены сотни.  
**2. В чём принципиальное отличие инвертированной таблицы от обычной многоуровневой таблицы?**  
Инвертированная таблица создаётся одна на все процессы и в ней количество строк равно количеству физических фреймов, а не количеству виртуальных страниц.  
**3. Почему x86-64 использует именно 9-битные индексы на каждом уровне PGD/PUD/PMD/PTE (а не, например, 10-битные)?**  
Архитектурно заложено, что таблица абсолютно любого уровня должна занимать ровно один физический фрейм (4096 байт=4КБ). Одна служебная запись (PTE) в 64-битной системе весит 8 байт. Посчитаем, сколько записей поместится в одну таблицу: 4096 / 8 = 512 записей. Чтобы процессор мог адресовать (пронумеровать) 512 строк, ему требуется ровно 9 бит данных (2⁹ = 512). Именно поэтому адрес бьется на порции по 9 бит  
**4. Что произойдёт при обходе таблиц страниц, если на уровне PMD обнаружен флаг \_PAGE_PSE? Почему в этом случае обход не продолжается до уровня PTE?**  
\_PAGE_PSE – Это специальный флаг (управляющий бит), который архитектура позволяет выставить на уровне таблицы PMD (3-й уровень). Когда аппаратный Page Walker процессора спускается на уровень PMD и видит в записи этот установленный флаг, его логика меняется. Он понимает, что данная запись указывает не на следующую таблицу (PTE), а напрямую на физический фрейм данных, но гигантского размера — 2 Мегабайта. Процессору больше не нужно читать таблицу PTE. Он сразу берет адрес из PMD, отбрасывает оставшиеся шаги трансляции и прибавляет к нему не 12 бит смещения, а 21 бит смещения (так как 2^(21) = 2  МиБ). Это колоссально экономит время на обход таблиц и разгружает кэш TLB.  
**5. Почему модуль pgtable_walk.ko обязан использовать mmap_read_lock() перед обходом mm_struct целевого процесса, а не может просто читать таблицы напрямую?**  
mmap_read_lock(mm) функция захватывает семафор (блокировку) на структуру памяти процесса в режиме чтения. Пока модуль держит эту блокировку, ядро Linux запрещает процессу изменять свою карту памяти (вызывать mmap, munmap, brk). Процесс «замораживается» в части изменения своих таблиц страниц, пока модуль безопасно не завершит обход. В чем опасность: Если процесс освободит память прямо в тот момент, когда модуль ядра читает его таблицу PTE, физические указатели в RAM станут невалидными. Модуль попытается прочесть несуществующую структуру, что приведет к Kernel Panic (синий экран смерти в Linux) и полному зависанию ОС.  
**6. Объясните, почему один и тот же виртуальный адрес общей переменной в главном потоке и в pthread-потоке одного процесса всегда транслируется в один и тот же физический кадр, а для двух процессов после fork — не обязательно, до и после операции записи в переменную.**  
Потоки (pthread): Все потоки внутри одного процесса разделяют общую память и используют одну и ту же структуру таблиц страниц (mm_struct). Поэтому один виртуальный адрес у любого потока всегда приведет к одному физическому кадру. Процессы после fork(): При создании нового процесса через fork() он получает копию таблиц страниц, работающую в режиме Copy-on-Write (COW).  
- До записи: Родитель и потомок используют одни и те же физические кадры в режиме Read-Only (чтобы экономить память).  
- После записи: Как только один из них пытается изменить переменную, процессор бьет COW fault, ядро Linux выделяет новый, отдельный физический кадр и копирует данные туда. Виртуальный адрес остается прежним, но физические кадры у процессов становятся разными.  
**7. Зачем понадобилась пятиуровневая схема (LA57/P4D) на некоторых современных процессорах?**  
При 4-уровневой схеме под виртуальный адрес отводится 48 бит. Математический предел адресного пространства, которое процессор может физически разметить, равен 2^(48)  байт = 256 Терабайт. Современным суперкомпьютерам и серверам баз данных этого стало не хватать. Архитектуру расширили до 57 бит виртуального адреса (режим Linear Address 57). Чтобы разметить такое гигантское пространство, 4 уровней не хватает — таблицы стали бы слишком большими. Поэтому между уровнями PGD и PUD добавили пятый уровень — P4D. Это подняло предел адресуемой памяти до 2^(57) байт = 128 Петабайт, что позволяет устанавливать в сервера терабайты физической RAM.
